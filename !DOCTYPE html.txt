let timerInterval;
let timeLeft = 0;
let studentInfo = {};

function startQuiz() {
    const name = document.getElementById("student-name").value.trim();
    const className = document.getElementById("class-name").value.trim();
    if (!name || !className) {
        alert("Vui lòng nhập đầy đủ Họ tên và Lớp!");
        return;
    }

    studentInfo = {
        name: name,
        className: className,
        lesson: document.getElementById("lesson-select").value
    };

    document.getElementById("login-card").classList.add("hidden");
    document.getElementById("quiz-card").classList.remove("hidden");

    renderQuiz();
    startTimer(quizData.timeMinutes * 60);
}

function startTimer(seconds) {
    timeLeft = seconds;
    updateTimerDisplay();
    timerInterval = setInterval(() => {
        timeLeft--;
        updateTimerDisplay();
        if (timeLeft <= 0) {
            clearInterval(timerInterval);
            alert("Hết giờ làm bài! Hệ thống sẽ tự động nộp bài.");
            submitQuiz();
        }
    }, 1000);
}

function updateTimerDisplay() {
    const m = Math.floor(timeLeft / 60).toString().padStart(2, '0');
    const s = (timeLeft % 60).toString().padStart(2, '0');
    document.getElementById("time-display").innerText = `${m}:${s}`;
}

function renderQuiz() {

    // Render Phần 1
    html += `<div class="section-title">PHẦN 1: Trắc nghiệm 4 lựa chọn (Chọn 1 đáp án đúng)</div>`;
    quizData.part1.forEach((q, idx) => {
        html += `<div class="question-box">
            <div class="question-title">${q.question}</div>
            <div class="options-grid">`;
        q.options.forEach((opt, optIdx) => {
            html += `<label class="option-item">
                <input type="radio" name="${q.id}" value="${optIdx}"> ${String.fromCharCode(65 + optIdx)}. ${opt}
            </label>`;
        });
        html += `</div></div>`;
    });

    // Render Phần 2
    html += `<div class="section-title">PHẦN 2: Trắc nghiệm Đúng/Sai</div>`;
    quizData.part2.forEach((q, idx) => {
        html += `<div class="question-box">
            <div class="question-title">${q.question}</div>
            <table class="tf-table">
                <thead><tr><th>Lựa chọn / Ý</th><th>Đúng</th><th>Sai</th></tr></thead>

        q.items.forEach((item, itemIdx) => {

        html += `</tbody></table></div>`;

        </div>`;
    });

    document.getElementById("quiz-content").innerHTML = html;
}

function submitQuiz() {
    clearInterval(timerInterval);

    let scoreP1 = 0, scoreP2 = 0, scoreP3 = 0;
    let details = { part1: [], part2: [], part3: [] };

    // Chấm Phần 1 (Mỗi câu 0.5 điểm)
    const p1Weight = 4 / quizData.part1.length;
    quizData.part1.forEach(q => {
        const selected = document.querySelector(`input[name="${q.id}"]:checked`);
        const userAns = selected ? parseInt(selected.value) : -1;
        const isCorrect = userAns === q.answer;
        if (isCorrect) scoreP1 += p1Weight;
        details.part1.push({ id: q.id, userAns, isCorrect });
    });

    // Chấm Phần 2 (Quy tắc thang điểm GDPT 2018: 1 ý=0.1, 2 ý=0.25, 3 ý=0.5, 4 ý=1.0)
    quizData.part2.forEach(q => {
        let correctCount = 0;
        let itemResults = [];
        q.items.forEach((item, itemIdx) => {
            const selected = document.querySelector(`input[name="${q.id}_${itemIdx}"]:checked`);
            const userAns = selected ? (selected.value === "true") : null;
            const isCorrect = userAns === item.answer;
            if (isCorrect) correctCount++;
            itemResults.push({ userAns, isCorrect });
        });

        let qScore = 0;
        if (correctCount === 1) qScore = 0.1;
        else if (correctCount === 2) qScore = 0.25;
        else if (correctCount === 3) qScore = 0.5;
        else if (correctCount === 4) qScore = 1.0;

        scoreP2 += qScore;
        details.part2.push({ id: q.id, correctCount, qScore, itemResults });
    });

    // Chấm Phần 3 (Mỗi câu 0.5 điểm)
    const p3Weight = 2 / quizData.part3.length;
    quizData.part3.forEach(q => {
        const input = document.querySelector(`input[name="${q.id}"]`);
        const userAns = input ? input.value.trim().toLowerCase() : "";
        const isCorrect = userAns === q.answer.toLowerCase();
        if (isCorrect) scoreP3 += p3Weight;
        details.part3.push({ id: q.id, userAns, isCorrect });
    });

    const totalScore = (scoreP1 + scoreP2 + scoreP3).toFixed(2);

    // Gửi kết quả về Google Sheets
    const payload = {
        studentName: studentInfo.name,
        className: studentInfo.className,
        lessonName: studentInfo.lesson,
        scoreP1: scoreP1.toFixed(2),
        scoreP2: scoreP2.toFixed(2),
        scoreP3: scoreP3.toFixed(2),
        totalScore: totalScore,
        details: details
    };

    if (SCRIPT_URL && SCRIPT_URL !== "https://script.google.com/macros/s/AKfycbw5hSL0y7OoGyPuY7a01n-nXb9HqI8FN9T8xBo_qJNcsZWoW92wEqghCNpbHZg1oGGYIQ/exec") {
        fetch(SCRIPT_URL, {
            method: "POST",
            mode: "no-cors",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify(payload)
        }).catch(err => console.error("Lỗi gửi điểm:", err));
    }

    // Hiển thị giao diện kết quả
    document.getElementById("quiz-card").classList.add("hidden");
    document.getElementById("result-card").classList.remove("hidden");

    document.getElementById("total-score-display").innerText = `${totalScore} / 10`;
    document.getElementById("sub-scores-display").innerText = 
        `Phần 1: ${scoreP1.toFixed(2)} đ | Phần 2: ${scoreP2.toFixed(2)} đ | Phần 3: ${scoreP3.toFixed(2)} đ`;

    renderReview();
}
